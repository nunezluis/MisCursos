\section{Mecánica clásica: el estudio del movimiento}
\subsection{¿Qué es la mecánica clásica?}
\frame{
  \frametitle{¿Qué es la mecánica clásica?}
   \begin{itemize}  
  	\item<1-> El estudio del \alert{movimiento}: la evolución en el tiempo de la posición de una partícula, o de la configuración de un sistema de partículas.
	%
	\item<2-> Válida en un amplio rango de escalas --- desde $\sim 100$\,nm hasta distancias cosmológicas --- excepto en el régimen cuántico.
	%
	\item<3-> ¿Dónde encuentra la mecánica clásica sus límites (siglo XX)?
	\begin{itemize}
		\item altas velocidades / energías $\;\rightarrow\;$ \emph{relatividad}
		\item escala atómica $\;\rightarrow\;$ \emph{mecánica cuántica}
	\end{itemize}	
    \end{itemize}
}

%%%%% Diapo 2
\subsection{Una dinámica, varios lenguajes}
\frame{
\frametitle{Una dinámica, varios lenguajes}
\begin{itemize}  
    \item<1-> {\bf Mecánica newtoniana}: fuerzas y aceleraciones \\ $\dt{\vek{p}} = \vek{F}, \qquad \vek{p}=m\dt{\vek{r}} .$
    \item<2-> {\bf Mecánica  lagrangiana}: Principio variacional 
    $\mathcal{L}(q_i,\dt q_i,t)=T-V,\quad \delta \mathcal{S}= \delta\int_{t_1}^{t_2} \mathcal{L}\,dt = 0\quad \quad \frac{d}{dt}\!\left(\pd{\mathcal{L}}{\dt q_i}\right)-\pd{\mathcal{L}}{q_i}=0.$ 
    \item<3-> {\bf Mecánica hamiltoniana}: flujo en el espacio de fase
    $\mathcal{H}(q_i,p_i,t)=\sum_i p_i\,\dt q_i - \mathcal{L}, \quad p_i=\pd{\mathcal{L}}{\dt q_i}.$
    \item<4->  Transformaciones canónicas:  la libertad de cambiar las variables del espacio de fase $(q_i,p_i)\;\longrightarrow\;(Q_i,P_i)$, preservando ec. de Hamilton
    \item<5->  Teoría de  Hamilton--Jacobi:  transformar un problema en su propia solución $ \mathcal{H}\!\left(q_i,\pd{\mathcal{S}}{q_i},t\right)+\pd{\mathcal{S}}{t}=0$,  $\mathcal{S}$ función principal de Hamilton 
    \item<6-> {\bf Para cada uno de estos formalismos,nos preguntaremos:} \emph{¿cómo resolvemos las ecuaciones?}, \emph{¿qué estructura le subyace?} Esto es simetrías, cantidades conservadas, integrabilidad.
    \item<7-> Haga que la IA resuelva un problema simple con todos estos métodos
 \end{itemize}
}

\section{Hoja de ruta del curso}
\frame{
\frametitle{Hoja de ruta del curso}
\  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{El formalismo, en orden}
      \begin{enumerate}\small
        \item Ecuaciones de movimiento; Lagrange (Cap.~1)
        \item Leyes de conservación y simetrí­as (Cap.~2)
        \item Fuerzas centrales, Kepler, dispersión (Cap.~3)
        \item Oscilaciones pequeñas, modos normales (Cap.~4)
        \item Cuerpos rí­gidos, tensor de inercia, Euler (Cap.~5)
        \item Dinámica hamiltoniana, transformaciones canónicas, H--J, integrabilidad (Cap.~6)
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Sistemas modelo recurrentes}
      \begin{itemize}\small
        \item oscilador armónico
        \item péndulo simple y no lineal
        \item problema de dos cuerpos / Kepler
        \item osciladores acoplados
        \item el trompo simétrico
      \end{itemize}
      \vspace{0.4em}
      {\footnotesize {\it Cada sistema se retoma en todos los formalismos, para comparar lo que revela cada lenguaje.}}
    \end{column}
  \end{columns}

      \begin{itemize}\small

    \end{column}
