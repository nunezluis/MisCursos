\section{Mecánica clásica: el estudio del movimiento}
\subsection{¿Qué es la mecánica clásica?}
\frame{
  \frametitle{¿Qué es la mecánica clásica?}
   \begin{itemize}  
  	\item<1-> El estudio del \alert{movimiento}: la evolución en el tiempo de la posición de una partícula, o de la configuración de un sistema de partículas.
	%
	\item<2-> Válida en un amplio rango de escalas --- desde $\sim 100$\,nm hasta distancias cosmológicas --- excepto en el régimen cuántico.
	%
	\item<3-> ¿Dónde encuentra la mecánica clásica sus límites (siglo XX)?
	\begin{itemize}
		\item altas velocidades / energías $\;\rightarrow\;$ \emph{relatividad}
		\item escala atómica $\;\rightarrow\;$ \emph{mecánica cuántica}
	\end{itemize}	
    \end{itemize}
}

%%%%% Diapo 2
\subsection{Una dinámica, varios lenguajes}
\frame{
\frametitle{Una dinámica, varios lenguajes}
\begin{itemize}  
    \item<1-> Mecánica newtoniana --- fuerzas y aceleraciones
    \item<2-> Mecánica  {lagrangiana} --- un principio variacional en el espacio de configuración
    \item<3-> Mecánica  {hamiltoniana} --- flujo en el espacio de fase
    \item<4->  {Transformaciones canónicas} --- la libertad de cambiar las variables del espacio de fase
    \item<5-> Teoría de  {Hamilton--Jacobi} --- transformar un problema en su propia solución
    \item<6-> Para cada uno de estos formalismos,nos preguntaremos: 
    \begin{itemize}
    	\item \emph{¿cómo resolvemos las ecuaciones?}
	\item \emph{¿qué estructura le subyace?} \\ Esto es simetrías, cantidades conservadas, integrabilidad.
    \end{itemize}
 \end{itemize}
}
